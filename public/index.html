    import { publishState } from './sync.js';const HEARTBEAT_MS = 5000;export function createPlayer(room, adapter) {  let currentId = room.currentId ?? null;  let status = 'idle';  const emit = (type, detail = {}) => {    publishState(room.id, { type, ...detail });  };  function selectNext() {    const playlist = room.playlist.filter(item => item.available);    if (playlist.length === 0) return null;    const currentIndex = playlist.findIndex(      item => item.id === currentId    );    const nextIndex = (currentIndex + 1) % playlist.length;    return playlist[nextIndex];  }  async function play(track) {    if (!track) return;    status = 'loading';    emit('track:loading', { trackId: track.id });    await adapter.load(track.videoId);    currentId = track.id;    status = 'playing';    emit('track:playing', { trackId: currentId });  }  async function onEnded() {    if (room.mode === 'repeat-one') {      await adapter.seekTo(0);      return adapter.play();    }    return play(selectNext());  }  const heartbeat = setInterval(() => {    if (status !== 'playing') return;    emit('playback:tick', {      trackId: currentId,      seconds: Math.floor(adapter.currentTime())    });  }, HEARTBEAT_MS);  adapter.on('ended', onEnded);  return {    play,    pause() {      status = 'paused';      adapter.pause();      emit('track:paused', { trackId: currentId });    },    next: () => play(selectNext()),    destroy() {      clearInterval(heartbeat);      adapter.off('ended', onEnded);    }  };}
